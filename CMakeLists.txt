cmake_minimum_required(VERSION 3.16)
project(openavc_hardware_example LANGUAGES C CXX)
set(OPENAVC_PRODUCT "${CMAKE_CURRENT_SOURCE_DIR}/product.json" CACHE FILEPATH "Equipment capability description")
option(OPENAVC_BUILD_EXAMPLES "Build the supplied amplifier examples" ON)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_library(http STATIC "${SDK}/civetweb/src/civetweb.c")
target_include_directories(http PUBLIC "${SDK}/civetweb/include")
target_compile_definitions(http PRIVATE NO_CGI NO_FILES NO_CACHING NO_SSL_DL OPENSSL_API_3_0 STOP_FLAG_NEEDS_LOCK)
target_link_libraries(http PUBLIC Threads::Threads OpenSSL::SSL OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_custom_command(
  OUTPUT "${GENERATED}/model.json" "${GENERATED}/model.provenance.json" "${GENERATED}/commands.c" "${GENERATED}/commands.h" "${GENERATED}/provenance.json"
  COMMAND "${Python3_EXECUTABLE}" "${SDK}/tools/prepare_product.py" --product "${OPENAVC_PRODUCT}" --core "${SDK}/contract" --out "${GENERATED}/model.json"
  COMMAND "${Python3_EXECUTABLE}" "${SDK}/tools/generate.py" --model "${GENERATED}/model.json" --core "${SDK}/contract" --out "${GENERATED}"
  DEPENDS "${OPENAVC_PRODUCT}" "${SDK}/tools/prepare_product.py" "${SDK}/tools/generate.py" "${SDK}/tools/state_mapping.py"
    "${SDK}/contract/openavc/drivers/avcdriver_semantic.py" "${SDK}/contract/openavc/drivers/spec.py" "${SDK}/contract/openavc/utils/regex_safety.py"
  VERBATIM)
add_library(openavc_runtime STATIC "${SDK}/src/endpoint.cpp" "${SDK}/src/adapter.cpp" "${SDK}/src/lifecycle.cpp")
target_compile_features(openavc_runtime PRIVATE cxx_std_17)
target_compile_options(openavc_runtime PRIVATE -Wall -Wextra -Werror)
target_include_directories(openavc_runtime PUBLIC "${SDK}/include" PRIVATE "${SDK}/json")
target_link_libraries(openavc_runtime PUBLIC http)
add_library(adapter STATIC "${SDK}/src/open_adapter.c" "${GENERATED}/commands.c")
target_compile_features(adapter PRIVATE c_std_11)
target_compile_options(adapter PRIVATE -Wall -Wextra -Werror)
target_include_directories(adapter PUBLIC "${SDK}/include" "${GENERATED}")
target_link_libraries(adapter PUBLIC openavc_runtime)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(openavc_runtime PRIVATE "${SDK}/src/bridge.cpp")
  add_executable(example-service "${SDK}/src/service.cpp")
  target_compile_features(example-service PRIVATE cxx_std_17)
  target_compile_options(example-service PRIVATE -Wall -Wextra -Werror)
  target_include_directories(example-service PRIVATE "${SDK}/json")
  target_link_libraries(example-service PRIVATE openavc_runtime)
else()
  target_sources(openavc_runtime PRIVATE "${SDK}/src/bridge_unavailable.cpp")
endif()
if(OPENAVC_BUILD_EXAMPLES)
foreach(language c cpp)
  add_executable(example-${language} "sample/example.${language}")
  target_compile_features(example-${language} PRIVATE c_std_11 cxx_std_17)
  target_compile_options(example-${language} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  target_link_libraries(example-${language} PRIVATE adapter)
endforeach()
endif()
