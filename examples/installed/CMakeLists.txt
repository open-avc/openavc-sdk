cmake_minimum_required(VERSION 3.16)
project(product_hardware LANGUAGES C CXX)
find_package(OpenAVC CONFIG REQUIRED)
set(OPENAVC_ADAPTER_SOURCE "" CACHE FILEPATH "Your product application implementing the generated callbacks")
if(NOT EXISTS "${OPENAVC_ADAPTER_SOURCE}")
  message(FATAL_ERROR "Set OPENAVC_ADAPTER_SOURCE to your hardware.c or hardware.cpp file.")
endif()
add_executable(hardware "${OPENAVC_ADAPTER_SOURCE}")
target_compile_features(hardware PRIVATE c_std_11 cxx_std_17)
target_link_libraries(hardware PRIVATE OpenAVC::adapter)
