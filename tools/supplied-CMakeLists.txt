cmake_minimum_required(VERSION 3.16)
project(openavc_manufacturer_product LANGUAGES C)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/build-target.json"
  CONTENT "{\"system\":\"${CMAKE_SYSTEM_NAME}\",\"machine\":\"${CMAKE_SYSTEM_PROCESSOR}\"}\n")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OPENAVC_PRODUCT "${CMAKE_CURRENT_SOURCE_DIR}/product.json" CACHE FILEPATH "Equipment capability description")
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(GLOB_RECURSE CONTRACT_SOURCES CONFIGURE_DEPENDS "${SDK}/contract/*.py")
add_custom_command(
  OUTPUT "${GENERATED}/model.json" "${GENERATED}/model.provenance.json" "${GENERATED}/commands.c" "${GENERATED}/commands.h" "${GENERATED}/provenance.json"
  COMMAND "${Python3_EXECUTABLE}" "${SDK}/tools/prepare_product.py" --product "${OPENAVC_PRODUCT}" --core "${SDK}/contract" --out "${GENERATED}/model.json"
  COMMAND "${Python3_EXECUTABLE}" "${SDK}/tools/generate.py" --model "${GENERATED}/model.json" --core "${SDK}/contract" --out "${GENERATED}"
  DEPENDS "${OPENAVC_PRODUCT}" "${SDK}/tools/prepare_product.py" "${SDK}/tools/generate.py" "${SDK}/tools/state_mapping.py" ${CONTRACT_SOURCES}
  VERBATIM)
add_library(oavc_runtime STATIC IMPORTED)
set_target_properties(oavc_runtime PROPERTIES IMPORTED_LOCATION "${SDK}/lib/libruntime.a")
add_library(oavc_http STATIC IMPORTED)
set_target_properties(oavc_http PROPERTIES IMPORTED_LOCATION "${SDK}/lib/libhttp.a")
set(OPENAVC_EXAMPLE_c "${CMAKE_CURRENT_SOURCE_DIR}/sample/example.c" CACHE FILEPATH "Product adapter source")
add_executable(example-c "${OPENAVC_EXAMPLE_c}" "${SDK}/src/open_adapter.c" "${GENERATED}/commands.c")
target_compile_features(example-c PRIVATE c_std_11)
target_compile_options(example-c PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_include_directories(example-c PRIVATE "${SDK}/include" "${GENERATED}")
target_link_libraries(example-c PRIVATE oavc_runtime oavc_http OpenSSL::SSL OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(example-c PRIVATE "${SDK}/lib/libc++.so.1" "${SDK}/lib/libc++abi.so.1" "${SDK}/lib/libunwind.so.1" m)
  file(RELATIVE_PATH SDK_LIB_FROM_BINARY "${CMAKE_CURRENT_BINARY_DIR}" "${SDK}/lib")
  set_target_properties(example-c PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/${SDK_LIB_FROM_BINARY}")
  add_custom_target(openavc-device-service
    COMMAND "${CMAKE_COMMAND}" -E create_symlink "${SDK}/bin/openavc-device-service" "${CMAKE_CURRENT_BINARY_DIR}/openavc-device-service"
    DEPENDS "${SDK}/bin/openavc-device-service"
    VERBATIM)
else()
  target_link_libraries(example-c PRIVATE c++)
endif()
